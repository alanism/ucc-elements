Enochian Elements

Instruments · signal, scale, and sequence

Enochian instruments for the atlas. These come from meters, tuners, turntables, tape decks and macro pads: hardware that shows its state through needles, lamps, key travel and light. Each one is translated into a pattern a web app can use.

Every element follows the atlas rules. Each has an Ivory and an Obsidian finish, one signal accent, spring motion, keyboard access, and a copyable implementation behind the Code Veil.

Same construction. No libraries and no hosted services. Springs, clicks and drawing are written inline, so any control can be lifted out on its own.

// VU scale is logarithmic: 0 VU sits at ~71% of the arc, +3 at 100%.
const vuPos = db => Math.pow(10, db / 20) / Math.pow(10, 3 / 20);
const vuAngle = p => -44 + p * 88;            // degrees either side of 12 o'clock

// Ballistics: an under-damped spring, integrated per frame.
const BALLISTIC = { k: 110, c: 15 };           // ≈300 ms rise, ~1.5% overshoot
function stepNeedle(n, targetDb, dt) {
  const a = -BALLISTIC.k * (n.p - vuPos(targetDb)) - BALLISTIC.c * n.v;
  n.v += a * dt;  n.p += n.v * dt;
  if (n.p > 1.04) { n.p = 1.04; n.v *= -0.3; }   // the pin stop bounces
  needle.setAttribute("transform", `rotate(${vuAngle(n.p)} 110 150)`);
}

// Every key is a request; the machine decides, and refusals explain themselves.
function press(k) {
  switch (k) {
    case "stop":  return go({ mode: "stop", rec: false, armed: false, pause: false });
    case "play":  return go({ mode: "play", rec: s.rec || s.armed, armed: false });
    case "pause": return s.mode === "play" ? go({ pause: !s.pause }) : refuse(k, "PAUSE NEEDS PLAY");
    case "rec":

      if (s.mode === "stop") return go({ armed: !s.armed });         // record-ready

// Every control folds into one 1024-step lookup table, so a fader move
// re-grades the frame with a single pass over the pixels.
function grade(x) {
  let v = x * 2 ** (exposure * 2);                              // ±2 stops
  v = (v + blacks * .1) / (1 - whites * .2 + blacks * .1);      // levels
  v += shadows * .18 * v * (1 - v) ** 2 * 6.75;                 // bump at ⅓
  v += highlights * .18 * v * v * (1 - v) * 6.75;               // bump at ⅔
  v = .5 + (v - .5) * (1 + contrast * .8);
  const p = clamp(v, 0, 1) * 7, i = Math.min(6, p | 0), f = p - i;
  v += lerp(zones[i], zones[i + 1], f * f * (3 - 2 * f)) * .12; // zone I…VIII
  return clamp(v, 0, 1);
}

// A compass needle is an under-damped pendulum on a magnetic spring.
const angDiff = (a, b) => ((a - b + 540) % 360) - 180;          // shortest signed turn
function needleStep(dt) {
